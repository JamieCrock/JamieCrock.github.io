Python from the analysis pipeline behind Analysing the Temporal Decay of AI Detectability: training the random forest detector over linguistic features with scikit-learn, fitting the decay model with NumPy, and producing publication figures with Matplotlib.

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.model_selection import StratifiedKFold


FEATURE_COLUMNS = [
    "type_token_ratio",
    "mean_sentence_length",
    "sentence_length_variance",
    "punctuation_density",
    "function_word_ratio",
    "mean_word_length",
    "burstiness",
]


def train_detector(df: pd.DataFrame, seed: int = 42) -> RandomForestClassifier:
    """Train the random forest detector on linguistic features."""
    X = df[FEATURE_COLUMNS]
    y = df["is_ai_generated"]

    model = RandomForestClassifier(
        n_estimators=500,
        max_depth=None,
        min_samples_leaf=2,
        class_weight="balanced",
        n_jobs=-1,
        random_state=seed,
    )

    cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=seed)
    auc_scores = []

    for train_idx, test_idx in cv.split(X, y):
        model.fit(X.iloc[train_idx], y.iloc[train_idx])
        proba = model.predict_proba(X.iloc[test_idx])[:, 1]
        auc_scores.append(roc_auc_score(y.iloc[test_idx], proba))

    print(f"cross-validated ROC AUC: {np.mean(auc_scores):.4f} "
          f"(+/- {np.std(auc_scores):.4f})")

    model.fit(X, y)
    print(classification_report(y, model.predict(X), digits=4))
    return model


def feature_importance(model: RandomForestClassifier) -> pd.Series:
    return pd.Series(
        model.feature_importances_, index=FEATURE_COLUMNS
    ).sort_values(ascending=False)
import numpy as np
import pandas as pd


def exponential_decay(t: np.ndarray, p0: float, lam: float, floor: float) -> np.ndarray:
    return floor + (p0 - floor) * np.exp(-lam * t)


def fit_decay(df: pd.DataFrame) -> dict:
    """Fit P(detect) = floor + (p0 - floor) * exp(-lam * t) by log-linear regression."""
    t = df["transformation_depth"].to_numpy(dtype=float)
    p = df["detection_rate"].to_numpy(dtype=float)

    floor = max(float(p.min()) - 0.02, 0.0)
    y = np.log(np.clip(p - floor, 1e-6, None))
    slope, intercept = np.polyfit(t, y, deg=1)

    lam = -slope
    p0 = floor + np.exp(intercept)
    fitted = exponential_decay(t, p0, lam, floor)

    ss_res = float(np.sum((p - fitted) ** 2))
    ss_tot = float(np.sum((p - p.mean()) ** 2))

    return {
        "p0": round(float(p0), 4),
        "lambda": round(float(lam), 4),
        "half_life": round(float(np.log(2) / lam), 2),
        "r_squared": round(1.0 - ss_res / ss_tot, 4),
    }


if __name__ == "__main__":
    observations = pd.read_csv("data/detection_rates.csv")
    for detector, group in observations.groupby("detector"):
        params = fit_decay(group)
        print(f"{detector}: half-life {params['half_life']} transformations, "
              f"R^2 {params['r_squared']}")
import matplotlib.pyplot as plt
import pandas as pd


def plot_decay(results: dict[str, pd.DataFrame]) -> None:
    """Plot detection rate against transformation depth for each detector."""
    fig, ax = plt.subplots(figsize=(8, 5))

    for detector, df in results.items():
        ax.plot(
            df["transformation_depth"],
            df["detection_rate"],
            marker="o",
            linewidth=1.8,
            label=detector,
        )

    ax.set_xlabel("Successive transformations")
    ax.set_ylabel("Detection rate")
    ax.set_title("Temporal decay of AI detectability by detector")
    ax.set_ylim(0.0, 1.0)
    ax.grid(alpha=0.25)
    ax.legend(frameon=False)

    fig.tight_layout()
    fig.savefig("figures/decay_by_detector.png", dpi=300)
    plt.close(fig)
